cmake_minimum_required(VERSION 3.24)

project(inference_runtime LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(INFERENCE_RUNTIME_ENABLE_TRITON_GRPC "Build with Triton gRPC client support" OFF)
option(INFERENCE_RUNTIME_ENABLE_LEGACY_ONNX "Build legacy ONNX Runtime sources" OFF)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

find_package(OpenCV REQUIRED COMPONENTS core videoio)
find_package(yaml-cpp REQUIRED)
find_package(PCRE2 CONFIG COMPONENTS 8BIT QUIET)

if(NOT TARGET PCRE2::8BIT)
  find_package(PCRE2 COMPONENTS 8BIT QUIET)
endif()

set(INFERENCE_RUNTIME_PCRE2_LINK_TARGET "")

if(DEFINED PCRE2_DIR)
  get_filename_component(INFERENCE_RUNTIME_PCRE2_PREFIX "${PCRE2_DIR}/../.." ABSOLUTE)
  set(INFERENCE_RUNTIME_PCRE2_INCLUDE_DIR "${INFERENCE_RUNTIME_PCRE2_PREFIX}/include")
  set(INFERENCE_RUNTIME_PCRE2_LIBRARY "${INFERENCE_RUNTIME_PCRE2_PREFIX}/lib/libpcre2-8.so")

  if(EXISTS "${INFERENCE_RUNTIME_PCRE2_INCLUDE_DIR}/pcre2.h" AND EXISTS "${INFERENCE_RUNTIME_PCRE2_LIBRARY}")
    add_library(InferenceRuntime::PCRE2_8BIT SHARED IMPORTED)
    set_target_properties(
      InferenceRuntime::PCRE2_8BIT
      PROPERTIES
        IMPORTED_LOCATION "${INFERENCE_RUNTIME_PCRE2_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${INFERENCE_RUNTIME_PCRE2_INCLUDE_DIR}"
    )
    set(INFERENCE_RUNTIME_PCRE2_LINK_TARGET InferenceRuntime::PCRE2_8BIT)
  endif()
endif()

if(NOT INFERENCE_RUNTIME_PCRE2_LINK_TARGET AND TARGET PCRE2::8BIT)
  set(INFERENCE_RUNTIME_PCRE2_LINK_TARGET PCRE2::8BIT)
elseif(NOT INFERENCE_RUNTIME_PCRE2_LINK_TARGET AND UNIX AND EXISTS "/usr/lib64/libpcre2-8.so.0")
  set(INFERENCE_RUNTIME_PCRE2_LINK_TARGET /usr/lib64/libpcre2-8.so.0)
endif()

if(TARGET yaml-cpp::yaml-cpp)
  get_target_property(INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION yaml-cpp::yaml-cpp IMPORTED_LOCATION)
  if(NOT INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION)
    get_target_property(INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION yaml-cpp::yaml-cpp IMPORTED_LOCATION_RELEASE)
  endif()

  if(INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION MATCHES "^/home/minhtk/miniconda3/")
    get_filename_component(INFERENCE_RUNTIME_YAML_LIB_NAME "${INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION}" NAME)
    get_filename_component(INFERENCE_RUNTIME_YAML_REAL_PATH "${INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION}" REALPATH)
    get_filename_component(INFERENCE_RUNTIME_YAML_REAL_NAME "${INFERENCE_RUNTIME_YAML_REAL_PATH}" NAME)
    set(INFERENCE_RUNTIME_RUNTIME_LIB_DIR "${CMAKE_BINARY_DIR}/runtime-libs")
    set(INFERENCE_RUNTIME_YAML_LOCAL_PATH "${INFERENCE_RUNTIME_RUNTIME_LIB_DIR}/${INFERENCE_RUNTIME_YAML_LIB_NAME}")
    get_target_property(INFERENCE_RUNTIME_YAML_INCLUDE_DIRS yaml-cpp::yaml-cpp INTERFACE_INCLUDE_DIRECTORIES)

    file(MAKE_DIRECTORY "${INFERENCE_RUNTIME_RUNTIME_LIB_DIR}")
    if(NOT EXISTS "${INFERENCE_RUNTIME_YAML_LOCAL_PATH}")
      file(CREATE_LINK "${INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION}" "${INFERENCE_RUNTIME_YAML_LOCAL_PATH}" SYMBOLIC COPY_ON_ERROR)
    endif()
    if(NOT EXISTS "${INFERENCE_RUNTIME_RUNTIME_LIB_DIR}/${INFERENCE_RUNTIME_YAML_REAL_NAME}")
      file(CREATE_LINK "${INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION}" "${INFERENCE_RUNTIME_RUNTIME_LIB_DIR}/${INFERENCE_RUNTIME_YAML_REAL_NAME}" SYMBOLIC COPY_ON_ERROR)
    endif()
    if(INFERENCE_RUNTIME_YAML_LIB_NAME MATCHES "^(libyaml-cpp\\.so\\.[0-9]+\\.[0-9]+)\\.[0-9]+$")
      set(INFERENCE_RUNTIME_YAML_SONAME "${CMAKE_MATCH_1}")
      if(NOT EXISTS "${INFERENCE_RUNTIME_RUNTIME_LIB_DIR}/${INFERENCE_RUNTIME_YAML_SONAME}")
        file(CREATE_LINK "${INFERENCE_RUNTIME_YAML_IMPORTED_LOCATION}" "${INFERENCE_RUNTIME_RUNTIME_LIB_DIR}/${INFERENCE_RUNTIME_YAML_SONAME}" SYMBOLIC COPY_ON_ERROR)
      endif()
    endif()

    add_library(InferenceRuntime::yaml-cpp SHARED IMPORTED)
    set_target_properties(
      InferenceRuntime::yaml-cpp
      PROPERTIES
        IMPORTED_LOCATION "${INFERENCE_RUNTIME_YAML_LOCAL_PATH}"
        INTERFACE_INCLUDE_DIRECTORIES "${INFERENCE_RUNTIME_YAML_INCLUDE_DIRS}"
    )
    set(INFERENCE_RUNTIME_YAML_TARGET InferenceRuntime::yaml-cpp)
  else()
    set(INFERENCE_RUNTIME_YAML_TARGET yaml-cpp::yaml-cpp)
  endif()
else()
  set(INFERENCE_RUNTIME_YAML_TARGET yaml-cpp)
endif()

if(INFERENCE_RUNTIME_ENABLE_TRITON_GRPC)
  include(FetchTritonGrpcClient)
endif()

add_executable(inference_runtime
  src/main.cpp
  src/infra/videosource.cpp
  src/infra/runtime_config_loader.cpp
  src/infra/triton_grpc_client.cpp
)

target_include_directories(inference_runtime
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/application/ports
    ${CMAKE_CURRENT_SOURCE_DIR}/src/application/dto
    ${CMAKE_CURRENT_SOURCE_DIR}/src/domain/entities
    ${CMAKE_CURRENT_SOURCE_DIR}/src/domain/value_object
    ${CMAKE_CURRENT_SOURCE_DIR}/src/infra
)

target_link_libraries(inference_runtime
  PRIVATE
    ${INFERENCE_RUNTIME_YAML_TARGET}
    ${OpenCV_LIBS}
)

if(INFERENCE_RUNTIME_PCRE2_LINK_TARGET)
  target_link_libraries(inference_runtime PRIVATE ${INFERENCE_RUNTIME_PCRE2_LINK_TARGET})
endif()

if(INFERENCE_RUNTIME_ENABLE_TRITON_GRPC)
  target_link_libraries(inference_runtime PRIVATE TritonClient::grpcclient)
  add_dependencies(inference_runtime triton_client_external)
  target_compile_definitions(inference_runtime PRIVATE INFERENCE_RUNTIME_HAS_TRITON_GRPC=1)
endif()

if(INFERENCE_RUNTIME_ENABLE_LEGACY_ONNX)
  message(WARNING "Legacy ONNX Runtime sources are not wired into the main target yet. Keep this OFF until include paths and ONNX Runtime packaging are cleaned up.")
endif()
